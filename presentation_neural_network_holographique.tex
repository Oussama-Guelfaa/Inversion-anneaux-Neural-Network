\section{Architecture du Réseau de Neurones}

\begin{frame}[fragile]{Architecture ImprovedDualParameterNet}
    \begin{columns}
        \begin{column}{0.6\textwidth}
            \textbf{Architecture Profonde Optimisée :}
            \begin{itemize}
                \item \textcolor{darkblue}{\textbf{Backbone partagé}} : 600→1024→768→512→256→128→64
                \item \textcolor{orange}{\textbf{Têtes spécialisées}} pour gap et L\_écran
                \item BatchNormalization + Dropout adaptatif
                \item Initialisation Xavier pour stabilité
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Innovations Clés :}
            \begin{itemize}
                \item Loss pondérée (gap×15, L\_écran×1)
                \item Dropout décroissant (0.15→0.06)
                \item Têtes asymétriques (gap plus complexe)
                \item Early stopping par fold
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{lstlisting}[caption=Têtes spécialisées]
# Tête Gap (plus complexe)
self.gap_head = nn.Sequential(
    nn.Linear(64, 32),
    nn.ReLU(),
    nn.Dropout(0.03),
    nn.Linear(32, 16),
    nn.ReLU(),
    nn.Linear(16, 1)
)

# Tête L_écran (simplifiée)
self.L_ecran_head = nn.Sequential(
    nn.Linear(64, 32),
    nn.ReLU(),
    nn.Dropout(0.03),
    nn.Linear(32, 1)
)
            \end{lstlisting}

\end{frame}

\begin{frame}[fragile]{Loss Function et Optimisation}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Loss Pondérée Personnalisée :}
            \begin{lstlisting}[caption=WeightedDualLoss]
class WeightedDualLoss(nn.Module):
    def __init__(self, gap_weight=15.0, 
                 L_ecran_weight=1.0):
        super().__init__()
        self.gap_weight = gap_weight
        self.L_ecran_weight = L_ecran_weight
        self.mse = nn.MSELoss()
    
    def forward(self, predictions, targets):
        gap_loss = self.mse(predictions[:, 0], 
                           targets[:, 0])
        L_ecran_loss = self.mse(predictions[:, 1], 
                               targets[:, 1])
        
        total_loss = (self.gap_weight * gap_loss + 
                     self.L_ecran_weight * L_ecran_loss)
        
        return total_loss, gap_loss, L_ecran_loss
            \end{lstlisting}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Configuration d'Entraînement :}
            \begin{itemize}
                \item \textbf{Optimiseur} : AdamW (lr=0.0005, wd=1e-4)
                \item \textbf{Scheduler} : ReduceLROnPlateau
                \item \textbf{Batch size} : 32
                \item \textbf{Early stopping} : patience=20
                \item \textbf{Gradient clipping} : max\_norm=1.0
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Justification du Poids Gap×15 :}
            \begin{itemize}
                \item Gap plus difficile à prédire
                \item Tolérance plus stricte (±0.01µm)
                \item Impact critique sur la précision
                \item Équilibrage des gradients
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

def run_kfold_validation(self, X, y, k_folds=5, epochs=200):
    """Exécute la validation croisée K-Fold."""

    # Mélange aléatoire des données (recommandation tuteur)
    indices = np.arange(len(X))
    np.random.shuffle(indices)
    X_shuffled = X[indices]
    y_shuffled = y[indices]

    # Initialiser K-Fold
    kfold = KFold(n_splits=k_folds, shuffle=False, random_state=None)

    fold_results = []
    fold_models = []

    # Exécuter chaque fold
    for fold_num, (train_idx, val_idx) in enumerate(kfold.split(X_shuffled)):
        X_train_fold = X_shuffled[train_idx]
        y_train_fold = y_shuffled[train_idx]
        X_val_fold = X_shuffled[val_idx]
        y_val_fold = y_shuffled[val_idx]

        # Entraîner le modèle pour ce fold
        model, history, metrics, scalers = self.train_single_fold(
            X_train_fold, y_train_fold, X_val_fold, y_val_fold,
            fold_num, epochs
        )

        fold_results.append(metrics)
        fold_models.append(model)

    return fold_results, fold_models, kfold_summary

def forward(self, x):

    features = self.dropout6(features)
